/** Fix probe: lesson S2 (ru-bins-astro) in six-DOF: the navigation's position error at the third stage's cut-off for each unit the heavy test flies, and the grade. */
import { it } from 'vitest';
import { writeFileSync } from 'node:fs';
import { Simulation } from '../../src/physics/simulation';
import { lessonConfig } from '../../src/lessons/config';
import { flightEnded, gradeLesson } from '../../src/lessons/grader';
import { MEASURES } from '../../src/lessons/measures';
import { packPath, readPackText } from '../../src/lessons/packs';
import type { Lesson } from '../../src/lessons/types';
import type { MissionState } from '../../src/config/mission-file';
import packText from '../../public/lessons/packs/ru-24-05-06.orbitlab-lesson.json?raw';
const PACK = readPackText(packText, packPath('ru-24-05-06'))!;
const l = PACK.lessons.find((x) => x.id === 'ru-bins-astro') as Lesson;
function fly(edit?: (s: MissionState) => void): Simulation {
  const sim = new Simulation(lessonConfig((l as never as { mission: never }).mission, edit), { headless: true });
  let guard = 0;
  while (!flightEnded(l, sim) && !sim.done && sim.state.t < 8000 && guard++ < 20_000_000) sim.step(sim.suggestedDt());
  return sim;
}
const navigation = (nav: object) => (s: MissionState) => { s.dynamics = { ...s.dynamics!, navigation: { ...s.dynamics!.navigation, ...nav } }; };
it('s2 nav', { timeout: 3_600_000 }, () => {
  const out: string[] = [];
  const only = process.env.ONLY?.split(',');
  const runs: [string, ((s: MissionState) => void) | undefined][] = [['as set', undefined], ['star tracker', navigation({ starTracker: true })],
    ['navigation grade', navigation({ grade: 'navigation' })], ['mems + star', navigation({ grade: 'mems', starTracker: true })]];
  for (const [name, edit] of runs) {
    if (only && !only.includes(name)) continue;
    const sim = fly(edit);
    const g = gradeLesson(l, sim);
    out.push(`${name}: error ${MEASURES['nav.positionError'].read(sim)?.toFixed(2)} m, verdict ${g.verdict}, t ${sim.state.t.toFixed(1)}, ${sim.events.filter((e) => /impact|abort|seco|parkingOrbit/.test(e.key)).map((e) => `${e.t.toFixed(1)} ${e.key}`).join(', ')}`);
    writeFileSync(process.env.PROBE_OUT ?? 's2nav.txt', out.join('\n') + '\n');
  }
});
